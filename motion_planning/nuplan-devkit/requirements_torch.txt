# torch==1.9.0+cu111/torchvision==0.10.0+cu111 are x86_64-only CUDA wheel
# builds; PyPI never published a +cu111 local build for Linux aarch64.
# The plain (non-CUDA-local-version) 1.9.0/0.10.0 releases do ship a
# manylinux2014_aarch64 wheel though, so aarch64 (ARM64 Linux servers, or
# an ARM64 Linux VM) gets those instead -- same exact torch/torchvision
# version as x86_64, just without the bundled CUDA build, since none of
# these VMs/servers have an aarch64 CUDA toolchain anyway.
torch==1.9.0+cu111;      platform_system == "Linux" and platform_machine != "aarch64"
torch==1.9.0;            platform_system == "Linux" and platform_machine == "aarch64"
torch==1.9.0;            platform_system == "Darwin"
# torch_scatter has a prebuilt wheel only for Linux x86_64 + this exact
# torch+cuda combo (served from the data.pyg.org find-links above), so on
# Linux x86_64 this line resolves to that wheel and needs no special
# handling. Everywhere else (aarch64, and likely macOS) there is no
# prebuilt wheel at all, so pip builds it from source -- and since pip's
# isolated build env can't see torch, that build fails with
# "ModuleNotFoundError: No module named 'torch'" unless torch is already
# installed AND this package is installed in its own separate pass with
# --no-build-isolation. See the aarch64 install sequence in the top-level
# README.md's Motion Planning Benchmark Setup section.
